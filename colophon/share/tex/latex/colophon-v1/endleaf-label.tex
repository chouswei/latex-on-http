% Spec segments (function rating current power extra package).
\def\EndleafSegJoin#1#2{\ifx#2\@empty\else\ifx\EndleafSpecText\@empty
  \def\EndleafSpecText{#2}\else\expandafter\def\expandafter\EndleafSpecText
  \expandafter{\EndleafSpecText\ #2}\fi\fi}
\def\EndleafBuildSpec{%
  \let\EndleafSpecText\@empty
  \EndleafSegJoin{f}{\EndleafSegF}\EndleafSegJoin{r}{\EndleafSegR}%
  \ifEndleafKeepCurrent\EndleafSegJoin{c}{\EndleafSegC}\fi
  \EndleafSegJoin{p}{\EndleafSegP}\EndleafSegJoin{e}{\EndleafSegE}%
  \ifEndleafKeepPackage\EndleafSegJoin{k}{\EndleafSegK}\fi}
\newif\ifEndleafKeepCurrent
\newif\ifEndleafKeepPackage
\newif\ifEndleafFits
% Width of #2 at size #1 pt; bold when \EndleafBold is \bfseries.
\def\EndleafMeasure#1#2{\setbox0=\hbox{\fontsize{#1pt}{#1pt}\selectfont\EndleafFace#2}\EndleafW=\wd0\relax}
% Body, written by the gate for each label:
%   \def\EndleafQrRects{\q{x1}{y1}{x2}{y2}...}
%   \EndleafStockLabel{modules}{name}{function}{rating}{current}{power}{extra}{package}{stockId}
\long\def\EndleafStockLabel#1#2#3#4#5#6#7#8#9{%
  \global\advance\EndleafLabelNo by 1
  \EndleafQrSide=#1\EndleafModule
  \EndleafColX=\dimexpr\EndleafLabelMX mm+\EndleafQrSide+\EndleafGap\relax
  \EndleafCol=\dimexpr\paperwidth-\EndleafLabelMX mm-\EndleafColX\relax
  % Name: 16 pt down to 9 pt in 0.5 pt steps.
  \let\EndleafFace\bfseries
  \def\EndleafSize{16}%
  \EndleafMeasure{\EndleafSize}{#2}%
  \loop
    \ifdim\EndleafW>\EndleafCol \ifdim\EndleafSize pt>9pt \EndleafFitstrue\else\EndleafFitsfalse\fi
    \else\EndleafFitsfalse\fi
  \ifEndleafFits
    \edef\EndleafSize{\EndleafPt{\dimexpr\EndleafSize pt-0.5pt\relax}}%
    \EndleafMeasure{\EndleafSize}{#2}%
  \repeat
  \ifdim\EndleafW>\EndleafCol \def\EndleafNameOk{over}\else\def\EndleafNameOk{ok}\fi
  \let\EndleafNameSize\EndleafSize
  \typeout{ENDLEAF_LABELFIT \the\EndleafLabelNo\space name \EndleafNameSize pt - \EndleafNameOk}%
  % Spec: 7 pt; drop package, then current; then 6.5 and 6 pt.
  \def\EndleafSegF{#3}\def\EndleafSegR{#4}\def\EndleafSegC{#5}%
  \def\EndleafSegP{#6}\def\EndleafSegE{#7}\def\EndleafSegK{#8}%
  \let\EndleafFace\mdseries
  \def\EndleafSize{7}\def\EndleafDropped{-}%
  \EndleafKeepCurrenttrue\EndleafKeepPackagetrue
  \EndleafBuildSpec\EndleafMeasure{\EndleafSize}{\EndleafSpecText}%
  \ifdim\EndleafW>\EndleafCol
    \ifx\relax#8\relax\else\def\EndleafDropped{package}\fi
    \EndleafKeepPackagefalse\EndleafBuildSpec\EndleafMeasure{\EndleafSize}{\EndleafSpecText}%
  \fi
  \ifdim\EndleafW>\EndleafCol
    \ifx\relax#5\relax\else
      \ifx\EndleafDropped\EndleafDash\def\EndleafDropped{current}\else\def\EndleafDropped{package,current}\fi
    \fi
    \EndleafKeepCurrentfalse\EndleafBuildSpec\EndleafMeasure{\EndleafSize}{\EndleafSpecText}%
  \fi
  \ifdim\EndleafW>\EndleafCol \def\EndleafSize{6.5}\EndleafMeasure{\EndleafSize}{\EndleafSpecText}\fi
  \ifdim\EndleafW>\EndleafCol \def\EndleafSize{6}\EndleafMeasure{\EndleafSize}{\EndleafSpecText}\fi
  \ifdim\EndleafW>\EndleafCol \def\EndleafSpecOk{over}\else\def\EndleafSpecOk{ok}\fi
  \let\EndleafSpecSize\EndleafSize
  \ifx\EndleafSpecText\@empty\else
    \typeout{ENDLEAF_LABELFIT \the\EndleafLabelNo\space spec \EndleafSpecSize pt \EndleafDropped\space \EndleafSpecOk}%
  \fi
  % Stock line: 6 pt.
  \EndleafMeasure{6}{Stock \##9}%
  \ifdim\EndleafW>\EndleafCol \def\EndleafStockOk{over}\else\def\EndleafStockOk{ok}\fi
  \typeout{ENDLEAF_LABELFIT \the\EndleafLabelNo\space stock 6pt - \EndleafStockOk}%
  \EndleafTop=\dimexpr\EndleafLabelMY mm+4\EndleafModule\relax
  \EndleafBottom=\dimexpr\EndleafLabelMY mm+\EndleafQrSide-4\EndleafModule\relax
  \noindent\begin{tikzpicture}[inner sep=0pt,outer sep=0pt]
    \useasboundingbox (0pt,0pt) rectangle (\paperwidth,-\paperheight);
    \begin{scope}[shift={(\EndleafLabelMX mm,-\EndleafLabelMY mm)},x=\EndleafModule,y=-\EndleafModule]
      \EndleafQrRects
    \end{scope}
    \node[anchor=north west,text width=\EndleafCol,align=left]
      at (\EndleafColX,-\EndleafTop)
      {\fontsize{\EndleafNameSize pt}{\EndleafNameSize pt}\selectfont\bfseries #2};
    \ifx\EndleafSpecText\@empty\else
    \node[anchor=west,text width=\EndleafCol,align=left]
      at (\EndleafColX,{-\EndleafLabelMY mm-0.5\EndleafQrSide})
      {\fontsize{\EndleafSpecSize pt}{\EndleafSpecSize pt}\selectfont\EndleafSpecText};
    \fi
    \node[anchor=south west,text width=\EndleafCol,align=left]
      at (\EndleafColX,-\EndleafBottom)
      {\fontsize{6pt}{6pt}\selectfont Stock \##9};
  \end{tikzpicture}%
  \par\newpage}
